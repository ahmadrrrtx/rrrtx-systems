# agent.py — Main RSS Intelligence Agent
# RRRTX SYSTEMS — Free Template
# Runs every 6h via cron, posts curated digest to Slack

import os
import sqlite3
import logging
from datetime import datetime, timezone
from pathlib import Path

from feeds import FEEDS, fetch_all_feeds
from classifier import classify_article, summarize_top
from slack import post_digest

# Setup logging — easy to tail when running in cron
LOG_DIR = Path("logs")
LOG_DIR.mkdir(exist_ok=True)
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.FileHandler(LOG_DIR / "agent.log"),
        logging.StreamHandler()
    ]
)
log = logging.getLogger("agent")

DB_PATH = "state.db"

def init_db():
    """Create table to track seen articles (prevents duplicate alerts)."""
    with sqlite3.connect(DB_PATH) as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS seen (
                url TEXT PRIMARY KEY,
                title TEXT,
                published TEXT,
                seen_at TEXT
            )
        """)
        conn.commit()

def is_seen(url: str) -> bool:
    with sqlite3.connect(DB_PATH) as conn:
        cur = conn.execute("SELECT 1 FROM seen WHERE url = ?", (url,))
        return cur.fetchone() is not None

def mark_seen(url: str, title: str, published: str):
    with sqlite3.connect(DB_PATH) as conn:
        conn.execute(
            "INSERT OR IGNORE INTO seen VALUES (?, ?, ?, ?)",
            (url, title, published, datetime.now(timezone.utc).isoformat())
        )
        conn.commit()

def run_cycle():
    """One full pipeline: fetch → filter → classify → summarize → post."""
    log.info("Starting cycle...")

    # 1. Fetch all feeds
    articles = fetch_all_feeds(FEEDS)
    log.info(f"Fetched {len(articles)} articles")

    # 2. Filter out already-seen
    fresh = [a for a in articles if not is_seen(a['url'])]
    log.info(f"{len(fresh)} new articles after dedup")

    if not fresh:
        log.info("Nothing new. Sleeping.")
        return

    # 3. Classify each via local LLM (signal vs noise)
    classified = []
    for article in fresh:
        score = classify_article(article)
        article['score'] = score
        classified.append(article)
        log.info(f"  [{score:.2f}] {article['title'][:60]}...")

    # 4. Summarize top 3
    top = sorted(classified, key=lambda x: x['score'], reverse=True)[:3]
    digest = summarize_top(top)

    # 5. Post to Slack
    post_digest(digest, top)

    # 6. Mark all as seen
    for article in fresh:
        mark_seen(article['url'], article['title'], article.get('published', ''))

    log.info(f"Cycle complete. Posted digest of {len(top)} articles.")

if __name__ == "__main__":
    init_db()
    run_cycle()

def fetch_one(feed: Dict) -> List[Dict]:
    """Fetch a single feed, return parsed articles. Failures are logged."""
    try:
        parsed = feedparser.parse(feed["url"])
        articles = []
        for entry in parsed.entries[:5]:  # top 5 per feed
            articles.append({
                "title": entry.get("title", ""),
                "url": entry.get("link", ""),
                "summary": entry.get("summary", "")[:500],
                "published": entry.get("published", ""),
                "category": feed["category"],
                "source": feed["url"],
            })
        return articles
    except Exception as e:
        print(f"[feeds] Failed: {feed['url']} — {e}")
        return []

def fetch_all_feeds(feeds: List[Dict]) -> List[Dict]:
    """Parallel fetch all feeds — fast even with 50+ sources."""
    all_articles = []
    with ThreadPoolExecutor(max_workers=10) as ex:
        results = list(ex.map(fetch_one, feeds))
    for articles in results:
        all_articles.extend(articles)
    return all_articles
classifier.py — Local LLM scoring
# classifier.py — Use Ollama to score articles locally
# Zero API cost. Runs entirely on your VPS.

import httpx
import json
import re

OLLAMA_URL = "http://localhost:11434"
MODEL = "gemma4:e4b"  # or "llama3.1:8b", "mistral:7b"

# The prompt that decides what's signal vs noise
# EDIT THIS to match what YOUR team cares about
SYSTEM_PROMPT = """You are an engineering intelligence filter.

Given an article, score it from 0.0 to 1.0 based on:
- Novelty (0.3): Is this new information or rehashed?
- Relevance (0.4): Is this useful for a senior engineer building AI/ecommerce systems?
- Actionability (0.3): Can the reader DO something with this today?

Respond with ONLY a JSON object: {"score": 0.0, "reason": "one sentence"}

Examples:
- New model release with benchmarks: 0.85
- Hiring announcement: 0.10
- Open-source tool launch: 0.75
- Generic "AI is changing everything" op-ed: 0.20
"""

def classify_article(article: dict) -> float:
    """Score an article 0.0-1.0 via local LLM."""
    user_msg = f"""Title: {article['title']}

Summary: {article['summary']}

Score this."""

    try:
        with httpx.Client(timeout=30.0) as client:
            resp = client.post(
                f"{OLLAMA_URL}/api/generate",
                json={
                    "model": MODEL,
                    "system": SYSTEM_PROMPT,
                    "prompt": user_msg,
                    "stream": False,
                    "options": {"temperature": 0.1},
                }
            )
            data = resp.json()
            text = data.get("response", "{}")

            # Extract JSON from response
            match = re.search(r'\{[^}]+\}', text)
            if match:
                result = json.loads(match.group())
                return float(result.get("score", 0.5))
    except Exception as e:
        print(f"[classifier] LLM error: {e}")
    return 0.5  # neutral on error

def summarize_top(articles: list) -> str:
    """One-paragraph summary of the top articles."""
    if not articles:
        return "No standout articles this cycle."

    prompt = f"""Summarize these {len(articles)} articles in 3 sentences,
each starting with the article number. Be specific and technical.

"""
    for i, a in enumerate(articles, 1):
        prompt += f"{i}. {a['title']}\n   {a['summary'][:200]}\n\n"

    try:
        with httpx.Client(timeout=60.0) as client:
            resp = client.post(
                f"{OLLAMA_URL}/api/generate",
                json={
                    "model": MODEL,
                    "prompt": prompt,
                    "stream": False,
                }
            )
            return resp.json().get("response", "Summary unavailable.").strip()
    except Exception as e:
        print(f"[summarize] error: {e}")
        return "Summary generation failed. See article links below."

# slack.py — Format & post digest to Slack via webhook
# Setup: Create incoming webhook at api.slack.com/messaging/webhooks

import os
import httpx
from datetime import datetime

WEBHOOK_URL = os.getenv("SLACK_WEBHOOK_URL")

def post_digest(summary: str, articles: list):
    """Build a clean Slack Block Kit message and POST it."""
    if not WEBHOOK_URL:
        print("[slack] No webhook configured — printing instead:")
        print(summary)
        for a in articles:
            print(f"  • {a['title']}\n    {a['url']}")
        return

    blocks = [
        {
            "type": "header",
            "text": {"type": "plain_text", "text": f"📡 Intelligence Digest — {datetime.now():%b %d, %H:%M}"}
        },
        {"type": "section", "text": {"type": "mrkdwn", "text": summary}},
        {"type": "divider"},
    ]

    for a in articles:
        blocks.append({
            "type": "section",
            "text": {
                "type": "mrkdwn",
                "text": f"*<{a['url']}|{a['title']}>*\n_Score: {a['score']:.2f} · {a['category']}_"
            }
        })

    try:
        with httpx.Client(timeout=15.0) as client:
            r = client.post(WEBHOOK_URL, json={"blocks": blocks})
            r.raise_for_status()
            print("[slack] Posted OK")
    except httpx.HTTPError as e:
        print(f"[slack] Failed: {e}")

#!/bin/bash
# setup.sh — Deploy this agent on a fresh Ubuntu VPS
# Tested on Hetzner CX22 ($4.50/mo) and DigitalOcean Basic ($6/mo)

set -e

echo "🚀 Setting up RSS Intelligence Agent..."

# 1. System deps
sudo apt update -qq
sudo apt install -y python3 python3-pip python3-venv cron

# 2. Install Ollama (local LLM runner)
if ! command -v ollama &> /dev/null; then
    curl -fsSL https://ollama.ai/install.sh | sh
fi

# 3. Pull the Gemma model (or swap for your preferred one)
echo "📥 Pulling Gemma 4 E4B model..."
ollama pull gemma4:e4b

# 4. Set up Python venv
python3 -m venv venv
source venv/bin/activate
pip install --quiet feedparser httpx

# 5. Set your Slack webhook (replace with your actual URL)
if [ ! -f .env ]; then
    echo "SLACK_WEBHOOK_URL=https://hooks.slack.com/services/YOUR/WEBHOOK/URL" > .env
    echo "⚠️  Edit .env with your real Slack webhook URL"
fi

# 6. Schedule to run every 6 hours via cron
echo "0 */6 * * * cd $(pwd) && ./venv/bin/python agent.py >> logs/cron.log 2>&1" | sudo crontab -

echo "✅ Done! Test with: ./venv/bin/python agent.py"
echo "📊 Check logs at: tail -f logs/agent.log"
echo "⏰ Runs every 6 hours via cron"
